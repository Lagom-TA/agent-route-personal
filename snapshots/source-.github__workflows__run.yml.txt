name: AgentRoute verified rules

on:
  workflow_dispatch:
    inputs:
      allow_shrink:
        description: Publish even if a dataset lost over 30 percent (only after checking upstream)
        type: boolean
        default: false
  schedule:
    # Beijing 06:30, matching the upstream schedule.
    - cron: '30 22 * * *'
  push:
    branches: [master]
  pull_request:

permissions:
  contents: read

concurrency:
  group: rules-${{ github.event_name }}-${{ github.ref }}
  cancel-in-progress: false

jobs:
  test:
    strategy:
      fail-fast: false
      matrix:
        os: [ubuntu-latest, macos-latest, windows-latest]
    runs-on: ${{ matrix.os }}
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
        with:
          persist-credentials: false
      - uses: actions/setup-python@ece7cb06caefa5fff74198d8649806c4678c61a1
        with:
          python-version: '3.11'
      - run: python -m pip install -r requirements-dev.txt
      - run: python -B -m unittest discover -s scripts -p "test_*.py"
      - run: python -O -B -m unittest discover -s scripts -p "test_*.py"
      - run: python -B scripts/check_contracts.py
  # Generate the text once, then compile optional representations from that snapshot.
  build:
    needs: test
    runs-on: ubuntu-latest
    timeout-minutes: 10
    outputs:
      artifact_id: ${{ steps.upload.outputs.artifact-id }}
      manifest_sha256: ${{ steps.candidate.outputs.manifest_sha256 }}
      release_baseline: ${{ steps.dataset.outputs.release_baseline }}
      stale_upstreams: ${{ steps.dataset.outputs.stale_upstreams }}
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
        with:
          persist-credentials: false
      - name: Build and validate the direct dataset
        id: dataset
        env:
          GITHUB_TOKEN: ${{ github.token }}
          AGENTROUTE_RELEASE_REPOSITORY: ${{ github.repository }}
          AGENTROUTE_ALLOW_SHRINK: ${{ inputs.allow_shrink }}
        run: python3 -B scripts/build_rules.py
      - name: Get pinned native compilers
        run: bash scripts/ci_rule_tools.sh
      - name: Compile optional formats from the same text generation
        id: candidate
        run: |
          python3 -B scripts/build_variants.py --source generated --output candidate \
            --mihomo "$RUNNER_TEMP/rule-tools/mihomo" --sing-box "$RUNNER_TEMP/rule-tools/sing-box"
          echo "manifest_sha256=$(sha256sum candidate/manifest.json | cut -d' ' -f1)" >> "$GITHUB_OUTPUT"
      - id: upload
        uses: actions/upload-artifact@cf430e030ddbb5b0abf93d22962f4752f3646cd9
        with:
          name: routing-data-${{ github.run_attempt }}
          path: candidate
          if-no-files-found: error
          retention-days: 7
  validate:
    needs: build
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
        with:
          persist-credentials: false
      - uses: actions/download-artifact@9000827ccba6bdab643e8b6fd33ac0654aef8333
        with:
          artifact-ids: ${{ needs.build.outputs.artifact_id || 'missing-artifact-id' }}
          path: generated
      - name: Validate the generated release contracts
        env:
          MANIFEST_SHA256: ${{ needs.build.outputs.manifest_sha256 }}
        run: |
          echo "$MANIFEST_SHA256  generated/manifest.json" | sha256sum --check --strict
          python3 -m pip install --disable-pip-version-check -r requirements-dev.txt
          python3 -B scripts/check_contracts.py --generated generated --require-variants
          python3 -B scripts/check_public_suffixes.py --generated generated
      - name: Verify with pinned official native clients
        run: |
          bash scripts/ci_rule_tools.sh
          python3 -B scripts/validate_native.py --mihomo "$RUNNER_TEMP/rule-tools/mihomo" --sing-box "$RUNNER_TEMP/rule-tools/sing-box"
      - name: Check sing-box DNS answers and selected outbounds on loopback
        run: |
          if ! command -v dig >/dev/null; then
            sudo apt-get update -qq
            sudo apt-get install -y --no-install-recommends bind9-dnsutils
          fi
          python3 -B scripts/check_sing_box_dns.py --sing-box "$RUNNER_TEMP/rule-tools/sing-box"
  # The only job with write access. It checks out no code and installs nothing:
  # it moves the validated bytes, and only the push step receives an explicit token.
  publish:
    needs: [build, validate]
    if: github.event_name != 'pull_request' && github.ref == 'refs/heads/master'
    permissions:
      contents: write
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/download-artifact@9000827ccba6bdab643e8b6fd33ac0654aef8333
        with:
          artifact-ids: ${{ needs.build.outputs.artifact_id || 'missing-artifact-id' }}
          path: publish
      - name: Prepare and verify the exact Git tree to publish
        id: prepare
        working-directory: publish
        shell: bash
        env:
          MANIFEST_SHA256: ${{ needs.build.outputs.manifest_sha256 }}
        run: |
          set -euo pipefail
          if [[ -n "$(find . ! -type f ! -type d -print)" ]]; then
            echo "::error::Release artifacts must contain only regular files and directories"
            exit 1
          fi
          if [[ "$(sha256sum manifest.json | cut -d' ' -f1)" != "$MANIFEST_SHA256" ]]; then
            echo "::error::Manifest differs from the build output"
            exit 1
          fi
          git init --quiet --initial-branch=direct-rules-release
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git -c core.autocrlf=false add --force --all
          git commit --quiet -m 'Publish one validated routing-data generation'
          git ls-tree -r --format='%(objectmode) %(path)' HEAD | LC_ALL=C sort > "$RUNNER_TEMP/release-tree"
          jq -r '((.files | keys[]), "manifest.json") | "100644 \(.)"' manifest.json | LC_ALL=C sort | diff "$RUNNER_TEMP/release-tree" -
          jq -r --arg manifest "$MANIFEST_SHA256" '.files + {"manifest.json": $manifest} | to_entries[] | [.key, .value] | @tsv' manifest.json |
            while IFS=$'\t' read -r path expected; do
              actual=$(git cat-file blob "HEAD:$path" | sha256sum | cut -d' ' -f1)
              if [[ "$actual" != "$expected" ]]; then
                echo "::error::Git blob differs from the build output: $path"
                exit 1
              fi
            done
      - name: Publish only the validated dataset
        id: push
        working-directory: publish
        shell: bash
        env:
          GITHUB_TOKEN: ${{ github.token }}
          RELEASE_BASELINE: ${{ needs.build.outputs.release_baseline }}
        run: |
          set -euo pipefail
          remote="https://x-access-token:${GITHUB_TOKEN}@github.com/${GITHUB_REPOSITORY}.git"
          if [[ "$RELEASE_BASELINE" == absent ]]; then
            expected=''
          elif [[ "$RELEASE_BASELINE" =~ ^[0-9a-f]{40}$ && ! "$RELEASE_BASELINE" =~ ^0{40}$ ]]; then
            expected="$RELEASE_BASELINE"
          else
            echo "::error::Missing or invalid release baseline from build"
            exit 1
          fi
          # Re-runs keep the original GITHUB_SHA. Only release is written by the push.
          if ! master_ref=$(git ls-remote --exit-code --refs "$remote" refs/heads/master); then
            echo "::error::Could not read the current master tip"
            exit 1
          fi
          if [[ "$master_ref" != "$GITHUB_SHA"$'\trefs/heads/master' ]]; then
            echo "::error::Original workflow source is no longer the master tip"
            exit 1
          fi
          git push "$remote" HEAD:refs/heads/release --force-with-lease="refs/heads/release:$expected"
  # The rules are already published. Failing here is the one signal that reaches the
  # maintainer: GitHub emails about failed runs and the build badge turns red.
  upstream-freshness:
    needs: [build, publish]
    if: needs.build.outputs.stale_upstreams != ''
    runs-on: ubuntu-latest
    timeout-minutes: 2
    steps:
      - name: Report upstream sources that stopped updating
        env:
          STALE_UPSTREAMS: ${{ needs.build.outputs.stale_upstreams }}
        run: |
          echo "::error::No new upstream commit in 14 days: $STALE_UPSTREAMS"
          exit 1
